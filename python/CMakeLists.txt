cmake_minimum_required(VERSION 3.26)
project(fastbdf_python LANGUAGES CXX)
find_package(Python 3.10 REQUIRED COMPONENTS Interpreter Development.Module OPTIONAL_COMPONENTS Development.SABIModule)
if(SKBUILD_SABI_COMPONENT STREQUAL "Development.SABIModule" AND NOT TARGET Python::SABIModule)
    message(FATAL_ERROR "An abi3 wheel requires Python Stable ABI development files")
endif()
find_package(nanobind CONFIG REQUIRED)
add_subdirectory(.. fastbdf-core EXCLUDE_FROM_ALL)
nanobind_add_module(fastbdf_python STABLE_ABI bindings.cpp)
target_link_libraries(fastbdf_python PRIVATE fastbdf::fastbdf)
set_target_properties(fastbdf_python PROPERTIES OUTPUT_NAME fastbdf)
install(TARGETS fastbdf_python LIBRARY DESTINATION . RUNTIME DESTINATION .)
